cmake_minimum_required(VERSION 3.20)
project(orb_nav_host C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS OFF)
enable_testing()
add_executable(test_nav ../orb_nav.c test_nav.c)
add_executable(test_nav_compositor ../orb_nav_compositor.c ../orb_nav_compositor_slots.c test_compositor.c)
add_executable(test_nav_iris_oracle ../orb_nav_compositor.c ../orb_nav_compositor_slots.c test_iris_oracle.c)
add_executable(test_nav_compositor_work ../orb_nav_compositor.c ../orb_nav_compositor_slots.c test_compositor_work.c)
target_compile_definitions(test_nav_compositor_work PRIVATE __XTENSA__ _FORTIFY_SOURCE=0 memcpy=orb_nav_test_memcpy)
target_compile_options(test_nav_compositor_work PRIVATE -fno-builtin-memcpy)
target_include_directories(test_nav_compositor_work PRIVATE ../../orb_lv_simd/include)
add_executable(test_nav_compositor_esp_allocator ../orb_nav_compositor.c ../orb_nav_compositor_slots.c test_compositor_esp_allocator.c)
target_compile_definitions(test_nav_compositor_esp_allocator PRIVATE ESP_PLATFORM)
target_include_directories(test_nav_compositor_esp_allocator PRIVATE stubs)
target_include_directories(test_nav_compositor_esp_allocator PRIVATE ../../orb_memory/include)
add_executable(test_nav_admission ../orb_nav_admission.c test_admission.c)
foreach(target test_nav test_nav_compositor test_nav_compositor_esp_allocator test_nav_admission test_nav_iris_oracle test_nav_compositor_work)
    target_include_directories(${target} PRIVATE ../include)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(${target} PRIVATE m)
    add_test(NAME ${target} COMMAND ${target})
endforeach()
